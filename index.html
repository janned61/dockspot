    <script>

        document.addEventListener('DOMContentLoaded', () => {

            loadBoatSpecs();

           

            // Sätt dagens datum som standard

            const today = new Date();

            const tomorrow = new Date(today);

            tomorrow.setDate(tomorrow.getDate() + 14); // Standard 2 veckor framåt

           

            document.getElementById('startDate').value = today.toISOString().split('T')[0];

            document.getElementById('endDate').value = tomorrow.toISOString().split('T')[0];

        });

 

        // Spara båtmått i localStorage

        function saveBoatSpecs() {

            const specs = {

                length: document.getElementById('length').value,

                width: document.getElementById('width').value,

                depth: document.getElementById('depth').value

            };

            localStorage.setItem('dockspot_boat_specs', JSON.stringify(specs));

           

            const tag = document.getElementById('savedTag');

            tag.style.display = 'inline';

            setTimeout(() => { tag.style.display = 'none'; }, 2000);

        }

 

        // Ladda sparade båtmått

        function loadBoatSpecs() {

            const saved = localStorage.getItem('dockspot_boat_specs');

            if (saved) {

                const specs = JSON.parse(saved);

                if (specs.length) document.getElementById('length').value = specs.length;

                if (specs.width) document.getElementById('width').value = specs.width;

                if (specs.depth) document.getElementById('depth').value = specs.depth;

            }

        }

 

        // Sök funktion som delar upp intervallet dygn för dygn

        async function runSearch() {

            const dock = document.getElementById('dockSelect').value;

            const length = document.getElementById('length').value;

            const width = document.getElementById('width').value;

            const depth = document.getElementById('depth').value;

           

            const startDateVal = document.getElementById('startDate').value;

            const endDateVal = document.getElementById('endDate').value;

 

            if (!startDateVal || !endDateVal) {

                alert('Vänligen välj både start- och slutdatum.');

                return;

            }

 

            const resultsContainer = document.getElementById('resultsContainer');

            resultsContainer.innerHTML = '<p style="font-size: 13px; text-align: center; color: var(--text-sub);">Hämtar tillgänglighet från Dockspot...</p>';

 

            let curr = new Date(startDateVal);

            const end = new Date(endDateVal);

            let htmlResults = '';

 

            // Loopa varje dygn i intervallet

            while (curr < end) {

                const arrival = curr.toISOString().split('T')[0];

               

                let nextDay = new Date(curr);

                nextDay.setDate(nextDay.getDate() + 1);

                const departure = nextDay.toISOString().split('T')[0];

 

                // Konstruera URL enligt Dockspots API-struktur

                const targetUrl = `https://www.dockspot.com/en/docks/${dock}/availability?length=${length}&width=${width}&depth=${depth}&arrival=${arrival}&departure=${departure}`;

               

                // Använd proxy för att kringgå CORS-begränsningar i webbläsaren

                const proxyUrl = `https://api.allorigins.win/get?url=${encodeURIComponent(targetUrl)}`;

 

                try {

                    const response = await fetch(proxyUrl);

                    const data = await response.json();

                   

                    // Kolla om vi fick giltigt svar

                    const isAvailable = data.contents && !data.contents.includes("No available spots");

 

                    htmlResults += `

                        <div class="day-result">

                            <div class="day-header">

                                <span>📅 ${arrival} ➔ ${departure}</span>

                                <span class="spot-badge ${isAvailable ? 'spot-available' : 'spot-full'}">

                                    ${isAvailable ? 'Plats Finns' : 'Fullbokat'}

                                </span>

                            </div>

                        </div>

                    `;

                } catch (err) {

                    htmlResults += `

                        <div class="day-result">

                            <div class="day-header">

                                <span>📅 ${arrival}</span>

                                <span class="spot-badge spot-full">Kunde inte hämta data</span>

                            </div>

                        </div>

                    `;

                }

 

                curr.setDate(curr.getDate() + 1);

            }

 

            resultsContainer.innerHTML = htmlResults;

        }

    </script>
